void avancer(distance){
    Setup de toutes les variables locales
    Lire la valeur des encodeurs de droite et gauche
    vitesse = Soustraire la valeur de l'encodeur réel à la valeur voulu * KP
    ajouter au compteur la valeur de l'encodeur
    distance = soustraire le compteur total réel à une multiplication de la valeur théorique * KI
    if compteur total réel <= valeur théorique:
        break
    modifier la vitesse du moteur selon (vitesse + distance)
    recommence pour l'autre moteur
    incrémenter le compteur de cycle
    delay (1000)
}
